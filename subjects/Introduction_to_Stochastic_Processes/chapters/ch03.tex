\section{Discrete Time Martingales}
\textit{September 16th.}

\begin{definition}
    A sequence of random variables $\{Z_{n}\}_{n \geq 1}$ is a \eax{martingale} if
    \begin{enumerate}
        \item $\E[\abs{Z_{n}}] < \infty$ for all $n \geq 1$,
        \item $\E[Z_{n} \mid Z_{n-1} = z_{n-1}, \ldots, Z_{1} = z_{1}] = z_{n-1}$.
    \end{enumerate}
\end{definition}
To understand the second condition, let us focus on the following: with $(\Omega,\cF,\Prob)$ as a probability space, and $Y,X:\Omega \to S$ be discrete random variables with $S$ a countable set, then with $y \in Y(\Omega)$, we have
\begin{align}
    \E[X \mid Y = y] = \sum_{k \in S} k \Prob(X = k \mid Y = y)
\end{align}
For multiple discrete random variables $Y_{1},Y_{2},\ldots,Y_{n}$, we have
\begin{align}
    \E[X \mid Y_{1}=y_{1},\ldots,Y_{n}=y_{n}] = \sum_{k \in S} k \Prob(X=k \mid Y_{1}=y_{1},\ldots,Y_{n}=y_{n})
\end{align}
for $y_{i} \in Y_{i}(\Omega)$.

\begin{example}
    Let $\{S_{n}\}_{n \geq 1}$ be the simple symmetric random walk on $\Z$, with $S_{0} = 0$, and let $s_{1},\ldots,s_{n-1} \in Z$ be such that $\Prob(S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}) > 0$. As a formal or imprecise calculation, we have
    \begin{align}
        \E[S_{n} \mid S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}] &= \sum_{k \in \Z} k \Prob(S_{n}=k \mid S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}) \notag \\
        &= \sum_{k =-n}^{n} k \Prob(S_{n-1}+X_{n}=k \mid S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}) \notag \\
        &= \sum_{k =-n}^{n} k \Prob(X_{n}=k-s_{n-1} \mid S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}) \notag \\
        &= \sum_{k =-n}^{n} k \Prob(X_{n}=k-s_{n-1}) \notag \\
        &= (s_{n-1}-1)\Prob(X_{n}=1) + (s_{n-1}+1)\Prob(X_{n}=-1) = s_{n-1}.
    \end{align}
    Also, $\E[\abs{S_{n}}] < \infty$ for all $n \geq 1$. If instead, we consider the simple asymmetric random walk on $\Z$ (that is, $\Prob(X_{n}=1) = p = 1-\Prob(X_{n}=-1)$ and $p \neq \frac{1}{2}$), then we have
    \begin{align}
        \E[S_{n} \mid S_{n-1}=s_{n-1},\ldots,S_{1}=s_{1}] &= (s_{n-1}-1)p + (s_{n-1}+1)(1-p) = s_{n-1} + 1 - 2p \neq s_{n-1}.
    \end{align}
    Hence, the simple symmetric random walk is a martingale, while the simple asymmetric random walk is not.
\end{example}

\begin{example}
    Let $X_{n}$ be a sequence of independent and identically distributed random variables with $\Prob(X_{n} = 2) = \frac{1}{2} = \Prob(X_{n} = 0)$, and then define $Z_{n} = \prod_{i=1}^{n} X_{i}$. Then, we have
    \begin{align}
        \E[\abs{Z_{n}}] = \prod_{k=1}^{n} \E[\abs{X_{k}}] = \prod_{k=1}^{n} 1 = 1 < \infty
    \end{align}
    for all $n \geq 1$. Now, for $z_{1},\ldots,z_{n-1} \in \{0,2,4,\ldots,2^{n-1}\}$ such that $\Prob(Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}) > 0$, we have
    \begin{align}
        \E[Z_{n} \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}] &= \E[X_{n} Z_{n-1} \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}] \notag \\
        &= z_{n-1} \E[X_{n} \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}] \notag \\
        &= z_{n-1} \E[X_{n}] = z_{n-1}.
    \end{align}
    Hence, $\{Z_{n}\}_{n \geq 1}$ is a martingale.
\end{example}

Suppose $\{Z_{n}\}_{n \geq 1}$ is a Markov chain. A consequence of the Markov property,
\begin{align}
    \Prob(Z_{n}=z_{n},\ldots,Z_{0}=z_{0}) = \mu(\{z_{0}\}) \prod_{k=1}^{n} p_{z_{k-1},z_{k}},
\end{align}
is that the past and the future are independent given the present. A martingale is less rigid than a Markov chain, saying that the expectation of the future given the present and the past is equal to the present. Moreover, Martingales are easier to construct and verify than Markov chains, and several limit results follow too.

\begin{example}
    Let $Z_{1}$ be the random variable defined by $\Prob(Z_{1}=1) = \frac{1}{2} = \Prob(Z_{1}=0)$. If $\{X_{n}\}_{n \geq 1}$ is a sequence of independent and identically distributed random variables with $\Prob(X_{n}=1) = \frac{1}{2} = \Prob(X_{n}=-1)$, then define $Z_{n} = Z_{n-1} + X_{n}Z_{1}$ for $n \geq 2$. Then, verify that $\{Z_{n}\}_{n \geq 1}$ is a Markov chain, but not a martingale.
\end{example}

\noindent From here on, we will interpet $\E[Z_{n} \mid z_{n-1},\ldots,z_{1}]$ to be $\E[Z_{n} \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}]$.

Let $\{Z_{n}\}_{n \geq 1}$ be a collection of discrete random variables on $(\Omega, \cF, \Prob)$. Then we know that
\begin{align}
    \E[Z_{n} \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}] = \sum_{k \in Z_{n}(\Omega)} k \Prob(Z_{n}=k \mid Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}).
\end{align}
Note that both sides of the above equation is a function of $z_{1},\ldots,z_{n-1}$, say, $f(z_{1},\ldots,z_{n-1})$. If $\{Z_{n}\}_{n \geq 1}$ were a martingale, then the equation above implies that $f(z_{1},\ldots,z_{n-1}) = z_{n-1}$, which is a function of $z_{n-1}$ alone. Thus, we define $Y_{n-1} \defeq f(z_{1},\ldots,z_{n-1})$, which we interpret as $\E[Z_{n} \mid z_{n-1},\ldots,z_{1}]$.

\begin{remark}
    We recall two important facts.
    \begin{enumerate}
        \item If $Y$ and $Z$ are discrete random variables, and $X = h(Y,Z)$ for some function $h:\R^{2}\to \R$, then $\E[X \mid Y=y,Z=z] = h(y,z)$.
        \item If $X$ and $Y$ are independent discrete random variables, then $\E[X \mid Y=y] = \E[X] \equiv \E[X \mid Y]$.
    \end{enumerate}
\end{remark}

\begin{example}
    Let $\{X_{n}\}_{n \geq 1}$ be a sequence of independent and identically distributed random variables with $\Prob(X_{n}=1) = \frac{1}{2} = \Prob(X_{n}=-1)$. Define $S_{n} = \sum_{i=1}^{n} X_{i}$ for $n \geq 1$, and $S_{0} = 0$. Now, define $\tilde{Z}_{n} = e^{rS_{n}}$ for all $n \geq 1$, where $r \in \R$ is a constant. Then, we have
    \begin{align}
        \E[\abs{\tilde{Z}_{n}}] = \E[e^{rS_{n}}] \leq c < \infty
    \end{align}
    as $\abs{S_{n}} \leq n$. Now, notice that $e^{rS_{n}} = e^{r\sum_{i=1}^{n} X_{i}} = e^{r\sum_{i=1}^{n} X_{i}}$. But with $\tilde{Y}_{i} = e^{rX_{i}}$, we have $\tilde{Y}_{1},\ldots,\tilde{Y}_{n}$ are independent and identically distributed random variables, and $\tilde{Z}_{n} = \prod_{i=1}^{n} \tilde{Y}_{i}$. This gives $\E[X_{i}] = \frac{e^{r}+e^{-r}}{2} \neq 1$ unless $r = 0$. Hence, $\tilde{Z}_{n}$ is not a martingale. To extract a martingale out of this, define
    \begin{align}
        q(r) = \log \E[\tilde{Y}_{i}] = \log \left( \frac{e^{r}+e^{-r}}{2} \right),
    \end{align}
    and let $Y_{i} = e^{rX_{i}-q(r)}$. Then, we have that $Y_{i}$'s are independent and identically distributed random variables with $\E[Y_{i}] = 1$. Now, define $Z_{n} = \prod_{i=1}^{n} Y_{i} = e^{rS_{n}-nq(r)}$. Then, we have, via the same argument as in \textbf{Example 3.3}, that $\{Z_{n}\}_{n \geq 1}$ is a (product) martingale. More generally, if $\{X_{n}\}_{n \geq 1}$ is a sequence of independent and identically distributed random variables, and $S_{n} = \sum_{i=1}^{n} X_{i}$, then $\{Z_{n}\}_{n \geq 1}$ defined by $Z_{n} = e^{rS_{n}-nq(r)}$ is a martingale, where $q(r) = \log \E[e^{rX_{1}}]$.
\end{example}

\begin{lemma}
    Let $\{Z_{n}\}_{n \geq 1}$ be a martingale. Then, for any $n \geq 1$, we have
    \begin{align}
        \E[Z_{n} \mid z_{i},z_{i-1},\ldots,z_{1}] = z_{i}
    \end{align}
    for all $1 \leq i \leq n$.
\end{lemma}
\begin{proof}
    We make use of the tower property of conditional expectation, which states that for any random variables $X,Y,Z$, we have
    \begin{align}
        \E[\E[X \mid Y,Z] \mid Y] = \E[X \mid Y].
    \end{align}
    We prove via induction. The base case is $n=1$, which is trivial. Now, assume the statement holds for $n = m$, and let $n = m+1$. Then,
    \begin{align}
        \E[Z_{m+1} \mid z_{i},z_{i-1},\ldots,z_{1}] &= \E[\E[Z_{m+1} \mid z_{m},z_{m-1},\ldots,z_{1}] \mid z_{i},z_{i-1},\ldots,z_{1}] \notag \\
        &= \E[Z_{m} \mid z_{i},z_{i-1},\ldots,z_{1}] = z_{i},
    \end{align}
    where the second equality follows from the martingale property, and the third equality follows from the induction hypothesis. Hence, by induction, the statement holds for all $n \geq 1$.
\end{proof}
If we apply the above lemma to $i = 1$, then
\begin{align}
    \E[Z_{n} \mid Z_{1}] = Z_{1} \implies \E[Z_{n}] = \E[\E[Z_{n} \mid Z_{1}]] = \E[Z_{1}],
\end{align}
for all $n \geq 1$. Hence, the expectation of a martingale is constant over time.

\begin{remark}
    We show the tower property of conditional expectation for discrete random variables. Let $X,Y,Z$ be discrete random variables, and let $y \in Y(\Omega)$ and $z \in Z(\Omega)$. Let
    \begin{align}
        h(y,z) = \E[X \mid Y=y,Z=z] = \sum_{k \in X(\Omega)} k \Prob(X=k \mid Y=y,Z=z).
    \end{align}
    Then,
    \begin{align}
        \E[h(Y,Z) \mid Y=y_{0}] &= \sum_{y \in Y(\Omega)} \sum_{z \in Z(\Omega)} h(y,z) \Prob(Y=y,Z=z \mid Y=y_{0}) = \sum_{z \in Z(\Omega)} h(y_{0},z) \Prob(Z=z \mid Y=y_{0}) \notag \\
        &= \sum_{z \in Z(\Omega)} \left( \sum_{k \in X(\Omega)} k \Prob(X=k \mid Y=y_{0},Z=z) \right) \Prob(Z=z \mid Y = y_{0}) \notag \\
        &= \sum_{z \in Z(\Omega)} \sum_{k \in X(\Omega)} k \cdot \frac{\Prob(X=k,Y=y_{0},Z=z)}{\Prob(Y=y_{0})} \notag \\
        &= \frac{1}{\Prob(Y=y_{0})}\sum_{z \in Z(\Omega)} \sum_{k \in X(\Omega)} k \Prob(X=k,Y=y_{0},Z=z) \notag \\
        &= \frac{1}{\Prob(Y=y_{0})}\sum_{k \in X(\Omega)} k \Prob(X=k,Y=y_{0}) = \E[X \mid Y=y_{0}].
    \end{align}
    Hence, $\E[h(Y,Z) \mid Y=y_{0}] = \E[X \mid Y=y_{0}]$, which is the tower property of conditional expectation.
\end{remark}

\begin{example}
    Suppose $\{X_{i}^{(n)}\}_{i,n\geq 1}$ is a collection of independent and identically distributed random variables with $\Prob(X_{i}^{(n)}=0) = \frac{1}{2} = \Prob(X_{i}^{(n)}=2)$, and define $Z_{1} = 1$ and $Z_{n} = \sum_{i=1}^{Z_{n-1}} X_{i}^{(n)}$ for $n \geq 2$. Each particle in generation $n-1$ produces a random number of particles in generation $n$, and the number of particles produced by each particle is independent and identically distributed. We have $\E[\abs{Z_{1}}] = 1 < \infty$, and
    \begin{align}
        \E[\abs{Z_{k+1}}] = \E\left[\abs{\sum_{i=1}^{Z_{k}} X_{i}^{(k+1)}}\right] \leq \E\left[\sum_{i=1}^{Z_{k}} \abs{X_{i}^{(k+1)}}\right] \leq 2\E[Z_{k}] < \infty
    \end{align}
    via induction. Now, assume $\Prob(Z_{n-1}=z_{n-1},\ldots,Z_{1}=z_{1}) > 0$. Then
    \begin{align}
        \E[Z_{n} \mid z_{n-1},\ldots,z_{1}] &= \E\left[\sum_{i=1}^{Z_{n-1}} X_{i}^{(n)} \mid z_{n-1},\ldots,z_{1}\right] = \E\left[\sum_{i=1}^{z_{n-1}} X_{i}^{(n)}\right] = z_{n-1} \E[X_{1}^{(n)}] = z_{n-1}.
    \end{align}
    Therefore, $\{Z_{n}\}_{n \geq 1}$ is a martingale. This is an example of a branching process.
\end{example}

\section{Stopped Processes}
Let $X:\Omega \to \R$ be a discrete random variable over a probability space $(\Omega,\cF,\Prob)$, and let $\cA_{X} = \{X^{-1}(B) \mid B \in \cB(\R)\}$ be the set of all events observable by $X$; the set of all events $E$ such that it may be written as the union of basic events $\{X = c\}$ with $c$ coming from $X(\Omega)$.

Suppose $Z_{1},\ldots,Z_{n}$ are discrete random variables. Let $\cA_{n}$ be the set of all events observable by $Z_{1},\ldots,Z_{n}$, or observable by time $k$; that is, the set of all events $E \subseteq \Omega$ such that it may be written as the union of basic events $\{Z_{1}=z_{1},\ldots,Z_{n-1}=z_{n-1}\}$ where $z_{i} \in Z_{i}(\Omega)$.

We will adopt the above notation.

\begin{definition}
    Let $\{Z_{n}\}_{n \geq 1}$ be a sequence of random variables. A map $T: \Omega \to \N \cup \{\infty\}$ is termed a \eax{stopping time} if $\{T=k\} \in \cA_{k}$, where $\cA_{k}$ is the set of all event observable by time $k$.
\end{definition}

\begin{definition}
    For any stopping time $T$ for a sequence $\{Z_{n}\}_{n \geq 1}$ of random variables, the \eax{stopped process} $Z_{n}^{T}$ is defined as
    \begin{align}
        Z_{n}^{T} = Z_{n \wedge T} \begin{cases}
            Z_{n} &\text{if } n < T,\\
            Z_{T} &\text{if } n \geq T.
        \end{cases}
    \end{align}
\end{definition}

\begin{theorem}
    Given a sequence $\{Z_{n}\}_{n \geq 1}$ of random variables and a stopping time $T$, the stopped process $\{Z_{n}^{T}\}_{n \geq 1}$ is a martingale whenever $\{Z_{n}\}_{n \geq 1}$ is a martingale.
\end{theorem}

\begin{proof}
    Note that we may write $Z_{n}^{T} = \sum_{i=1}^{n-1} Z_{i} \1_{\{T=i\}} + Z_{n} \1_{\{T \geq n\}}$. Then, we have
    \begin{align}
        \E[\abs{Z_{n}^{T}}] \leq \sum_{k=1}^{n} \E[\abs{Z_{k}}] < \infty
    \end{align}
    as $\{Z_{n}\}_{n \geq 1}$ is a martingale. Moreover,
    \begin{align}
        \E[Z_{n}^{T} \mid \cA_{n-1}] &= \E[\sum_{i=1}^{n-1} Z_{i} \1_{\{T=i\}} \mid \cA_{n-1}] + \E[Z_{n} \1_{T \geq n} \mid \cA_{n-1}] = \sum_{i=1}^{n-1} Z_{i} \1_{\{T=i\}} + \1_{T \geq n} \E[Z_{n} \mid \cA_{n-1}] \notag \\
        &= \sum_{i=1}^{n-1} Z_{i} \1_{\{T=i\}} + \1_{T \geq n} Z_{n-1} = \sum_{i=1}^{n-2} Z_{i} \1_{\{T=i\}} + \1_{T \geq n-1} Z_{n-1} = Z_{n-1}^{T}.
    \end{align}
    This, however, does not show that the stopped process is a martingale, as we need to show that $\E[Z_{n}^{T} \mid z_{n-1}^{T},\ldots,z_{1}^{T}] = z_{n-1}^{T}$ for all $n \geq 1$. This is the same as $\E[Z_{n}^{T} \mid \cB_{n-1}]$, where $\cB_{n-1}$ are the events observable by $Z_{1}^{T},\ldots,Z_{n-1}^{T}$, which may not be the same as $\cA_{n-1}$. However, since $\cB_{n-1} \subseteq \cA_{n-1}$, we have
    \begin{align}
        \E[Z_{n}^{T} \mid \cB_{n-1}] = \E[\E[Z_{n}^{T} \mid \cA_{n-1}] \mid \cB_{n-1}] = Z_{n-1}^{T}.
    \end{align}
\end{proof}

\begin{corollary}
    If $T$ is a stopping time for a martingale $\{Z_{n}\}_{n \geq 1}$, then $\E[Z_{n}^{T}] = \E[Z_{1}]$.
\end{corollary}
\begin{proof}
    From above, $\{Z_{n}^{T}\}_{n \geq 1}$ is a martingale, and hence $\E[Z_{n}^{T}] = \E[Z_{1}^{T}] = \E[Z_{1}]$.
\end{proof}

Note that $Z_{n}^{T} \xrightarrow{n \to \infty} Z_{T}$ with probability $1$, and that $\E[Z_{n}^{T}] = \E[Z_{1}]$. Does this imply that $\E[Z_{n}^{T}] \xrightarrow{n \to \infty} \E[Z_{T}]$? The following is a counterexample to this claim.

\begin{example}
    Let $\{Z_{n}\}_{n \geq 1}$ be the product martingale sequence; that is, $\Prob(Z_{n} = 2^{n}) = \frac{1}{2^{n}}$ and $\Prob(Z_{n}=0) = 1-\frac{1}{2^{n}}$. Let $T = \min\{n \geq 1 \mid Z_{n} = 0\}$ be a stopping time. Then $Z_{T} = 0$ with probability $1$, so $\E[Z_{T}] = 0$. However, $\E[Z_{n}^{T}] = \E[Z_{1}] = 1$. Thus, $Z_{n}^{T} \xrightarrow{n \to \infty} Z_{T}$ with probability $1$, but $\E[Z_{n}^{T}]$ does not converge to $\E[Z_{T}]$.
\end{example}

\begin{theorem}[The \eax{optimal stopping theorem}]
    Let $T$ be a stopping time for a martingale $\{Z_{n}\}_{n \geq 1}$. Then $\E[Z_{T}] = \E[Z_{1}]$ if and only if $\E[\abs{Z_{T}}] < \infty$ and $\E[Z_{n} \mid T \geq n] \cdot \Prob(T \geq n) \xrightarrow{n \to \infty} 0$.
\end{theorem}

\begin{example}
    Consider $\{S_{n}\}_{n \geq 1}$ to be the simple asymmetric random walk on $\Z$, with step probabilities $\Prob(X_{i}=1) = p = 1-\Prob(X_{i}=-1)$. Then the sequence $\{W_{n}\}_{n \geq 1}$ defined by $W_{n} = \left( \frac{1-p}{p} \right)^{S_{n}}$ is a martingale. Now, for $a < 0 < b$, consider the stopping times $\sigma_{a} = \min\{n \geq 1 \mid S_{n}=a\}$ and $\sigma_{b} = \min\{n \geq 1 \mid S_{n}=b\}$, and let $T = \sigma_{a} \wedge \sigma_{b}$, a stopping time for $\{W_{n}\}_{n \geq 1}$. If the conditions for the optimal stopping theorem holds, then $\E[W_{T}] = \E[W_{1}]$. The left term is
    \begin{align}
        \E[W_{\sigma_{a}} \mid \sigma_{a} < \sigma_{b}] + \E[W_{\sigma_{b}} \mid \sigma_{a} \geq \sigma_{b}] = \left( \frac{1-p}{p} \right)^{a} \Prob(\sigma_{a} < \sigma_{b}) + \left( \frac{1-p}{p} \right)^{b} \Prob(\sigma_{a} \geq \sigma_{b}),
    \end{align}
    and the right term is $\E[W_{1}] = 1$. Thus, $\E[W_{T}] = 1$. Solving this alongside $\Prob(\sigma_{a} < \sigma_{b}) + \Prob(\sigma_{b} \leq \sigma_{a}) = 1$ gives
    \begin{align}
        \Prob(\sigma_{a} < \sigma_{b}) = \dfrac{\left( \frac{1-p}{p} \right)^{b} - 1}{\left( \frac{1-p}{p} \right)^{b} - \left( \frac{1-p}{p} \right)^{a}}
    \end{align}
    for $a < 0 < b$. This is the probability that the simple asymmetric random walk hits $a$ before hitting $b$. We now show that the conditions for the optimal stopping theorem holds. First, we have
    \begin{align}
        \E[\abs{W_{T}}] = \E[W_{T}] < \infty
    \end{align}
    since $W_{T}$ can only take finitely many values. Also, since $\Prob(T \geq n) \to 0$ as $n \to \infty$, and $\E[W_{n} \mid T \geq n]$ is bounded, we have $\E[W_{n} \mid T \geq n] \cdot \Prob(T \geq n) \to 0$ as $n \to \infty$. Hence, the conditions for the optimal stopping theorem holds.
\end{example}

\begin{proof}[Proof of the optimal stopping theorem.]
    Since $\{Z_{n}^{T}\}_{n \geq 1}$ is a martingale, we have $\E[\abs{Z_{n}^{T}}] < \infty$ and $\E[Z_{n}^{T}] = \E[Z_{1}]$ for all $n \geq 1$, and $Z_{n}^{T} \xrightarrow{n \to \infty} Z_{T}$ with probability $1$. Now,
    \begin{align}
        \E[Z_{n}^{T}] &= \sum_{i=1}^{n-1} \E[Z_{n}^{T}; T=i] + \E[Z_{n}^{T}; T \geq n] = \sum_{i=1}^{n-1} \E[Z_{i} \mid T=i] \cdot \Prob(T=i) + \E[Z_{n} \mid T \geq n] \cdot \Prob(T \geq n) \notag \\
        &= \sum_{i=1}^{n-1} \E[Z_{i} \mid T=i] \cdot \Prob(T=i) + \E[Z_{n} \mid T \geq n] \cdot \Prob(T \geq n).
    \end{align}
    Also, $\E[Z_{T}] = \sum_{i=1}^{\infty} \E[Z_{i} \mid T=i] \cdot \Prob(T=i)$ if and only if $\E[\abs{Z_{T}}] < \infty$. Hence, $\E[Z_{n}^{T}] \xrightarrow{n \to \infty} \E[Z_{T}]$ if and only if $\E[\abs{Z_{T}}] < \infty$ and $\E[Z_{n} \mid T \geq n] \cdot \Prob(T \geq n) \xrightarrow{n \to \infty} 0$. Since $\E[Z_{n}^{T}] = \E[Z_{1}]$ for all $n$, we have $\E[Z_{T}] = \E[Z_{1}]$ if and only if the two conditions above hold.
\end{proof}

The following serves as a process of checking the conditions for the optimal stopping theorem.
\begin{enumerate}
    \item If $\Prob(T < \infty) = 1$ (that is, $\Prob(T \geq n) \to 0$ as $n \to \infty$), and $\{Z_{n}\}_{n \geq 1}$ is a bounded martingale.
    \item $\E[T] < \infty$ and there exists $c > 0$ such that $\abs{Z_{n} - Z_{n-1}} < c$ for all $n \geq 2$.
    \item $T:\Omega \to \N$, and $\sup_{\omega \in \Omega} T(\omega) < \infty$.
\end{enumerate}

\section{Kolmogorov's Maximal Inequalities}
\textit{October 7th.}

Recall the inequalities of Markov and Chebyshev. Let $X$ be a non-negative random variable, and let $a > 0$. Then, we have
\begin{align}
    \Prob(X > a) \leq \frac{\E[X]}{a}, \qquad \Prob(\abs{X-\E[X]} \geq a) \leq \frac{\Var(X)}{a^{2}}.
\end{align}

\begin{theorem}[\eax{Kolmogorov's maximal inequality}]
    Let $\{Z_{n}\}_{n \geq 1}$ be a martingale, with $\E[Z_{n}^{2}] < \infty$ for all $n \geq 1$. Then, for any all $n \geq 1$ and $b > 0$, we have
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq n} \abs{Z_{i}} > b \right) \leq \frac{\E[Z_{n}^{2}]}{b^{2}}.
    \end{align}
\end{theorem}

If $h:\R \to \R$ is a convex function, then by Jensen's inequality, we have
\begin{align}
    \E[h(Z_{n}) \mid z_{n-1},\ldots,z_{1}] \geq h(\E[Z_{n} \mid z_{n-1},\ldots,z_{1}]) = h(z_{n-1}),
\end{align}
which is the property of a submartingale.
\begin{definition}
    A sequence of random variables $\{Z_{n}\}_{n \geq 1}$ is a \eax{submartingale} if $\E[\abs{Z_{n}}] < \infty$ for all $n \geq 1$, and
    \begin{align}
        \E[Z_{n} \mid z_{n-1},\ldots,z_{1}] \geq z_{n-1}
    \end{align}
\end{definition}
As above, if $\{Z_{n}\}_{n \geq 1}$ is a martingale, then $\{h(Z_{n})\}_{n \geq 1}$ is a submartingale for any convex function $h:\R \to \R$. In particular, $\{Z_{n}^{2}\}_{n \geq 1}$, $\{\abs{Z_{n}}\}_{n \geq 1}$, and $\{\abs{Z_{n}}\}_{n \geq 1}$ are submartingales.

\begin{proposition}
    Let $\{Z_{n}\}_{n \geq 1}$ be a non-negative submartingale. Then for any $n \geq 1$ and $a > 0$, we have
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq n} Z_{i} > a \right) \leq \frac{\E[Z_{n}]}{a}.
    \end{align}
\end{proposition}
We shall first use this proposition to prove Kolmogorov's maximal inequality.
\begin{proof}[Proof of Kolmogorov's maximal inequality]
    Suppose $\{Z_{n}\}_{n \geq 1}$ is a martingale with all finite second moments. Then $\{Z_{n}^{2}\}_{n \geq 1}$ is a submartingale, and the above proposition gives
    \begin{align}
        \Prob\left(\max_{1 \leq i \leq n} \abs{Z_{i}} > a\right) = \Prob\left(\max_{1 \leq i \leq n} Z_{i}^{2} > a^{2}\right) \leq \frac{\E[Z_{n}]^{2}}{a^{2}}.
    \end{align}
\end{proof}

\begin{proof}
    Let $J = \min\{k \geq 1 \mid Z_{k} \geq a\} \wedge m$ for some $m \geq 1$. Then $Z_{J} \geq a$ holds if and only if $Z_{n} \geq a$ for all $n \geq m$. Then, we have
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq m} Z_{i} > a \right) = \Prob(Z_{J} \geq a) &\leq \frac{\E[Z_{J}]}{a} = \frac{1}{a} \sum_{k=1}^{m-1} \E[Z_{k} \1_{\{J=k\}}] + \frac{1}{a} \E[Z_{m} \1_{\{J=m\}}] \notag \\
        &= \frac{1}{a} \sum_{k=1}^{m-1} \E[\E[Z_{m} \1_{\{J=k\}} \mid \cA_{k}]] + \frac{1}{a} \E[Z_{m} \1_{\{J=m\}}] \notag \\
        &= \frac{1}{a} \sum_{k=1}^{m-1} \E[\1_{\{J=k\}} \E[Z_{m} \mid \cA_{k}]] + \frac{1}{a} \E[Z_{m} \1_{\{J=m\}}] \notag \\
        &\leq \frac{1}{a} \sum_{k=1}^{m-1} \E[\1_{\{J=k\}} Z_{k}] + \frac{1}{a} \E[Z_{m} \1_{\{J=m\}}] = \frac{1}{a} \sum_{k=1}^{m} \E[Z_{k} \1_{\{J=k\}}] = \frac{\E[Z_{m}]}{a}.
    \end{align}
\end{proof}

\begin{proposition}
    Let $\{Z_{n}\}_{n \geq 1}$ be a non-negative martingale. Then
    \begin{align}
        \Prob\left( \sup_{n \geq 1} Z_{n} > a \right) \leq \frac{\E[Z_{1}]}{a}
    \end{align}
    for all $a > 0$.
\end{proposition}

\begin{proof}
    For $m \geq 1$, define $A_{m} = \{\max_{1 \leq i \leq m} Z_{i} > a\} = \bigcup_{i=1}^{m} \{Z_{i} > a\}$. Then, $A_{m} \subseteq A_{m+1}$; but note that $\bigcup_{m=1}^{\infty} A_{m} = \bigcup_{n=1}^{\infty} \{Z_{n} > a\}$ is \textit{not} the same as $\{\sup_{n \geq 1} Z_{n} > a\}$, as the latter is the event that $Z_{n} > a$ for infinitely many $n$, and the former is the event that $Z_{n} > a$ for some $n$. Let $\eps_{0} \in (0,a)$. Then, we have
    \begin{align}
        \Prob\left( \sup_{i \geq 1} Z_{i} \geq a \right) = \Prob\left( \bigcap_{\eps > 0} \bigcup_{i \geq 1} \{Z_{i} \geq a - \eps\} \right) \leq \Prob(\bigcup_{i \geq 1} \{Z_{i} \geq a - \eps_{0}\}).
    \end{align}
    Let $b = a-\eps_{0}$. By the previous proposition, we have
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq m} Z_{i} \geq b \right) \leq \frac{\E[Z_{1}]}{b}.
    \end{align}
    Taking the limit as $m \to \infty$ gives
    \begin{align}
        \Prob\left( \bigcup_{i=1}^{\infty} \{Z_{i} \geq b\} \right) \leq \frac{\E[Z_{1}]}{b}.
    \end{align}
    Combining the above inequalities gives
    \begin{align}
        \Prob\left( \sup_{i \geq 1} Z_{i} \geq a \right) \leq \frac{\E[Z_{1}]}{b} = \frac{\E[Z_{1}]}{a-\eps_{0}}.
    \end{align}
    Letting $\eps_{0} \to 0$ gives the desired result.
\end{proof}

\begin{theorem}[Convergence theorem]
    Let $\{Z_{n}\}_{n \geq 1}$ be a martingale.
    \begin{enumerate}
        \item If $Z_{n} \geq 0$ for all $n \geq 1$, then there exists a random variable $Z$ such that $Z_{n} \xrightarrow{n \to \infty} Z$ with probability $1$.
        \item If $\sup_{n \geq 1} \E[\abs{Z_{n}}] < \infty$, then there exists a random variable $Z$ such that $Z_{n} \xrightarrow{n \to \infty} Z$ with probability $1$.
        \item If $\sup_{n \geq 1} \E[Z_{n}^{2}] < \infty$, then there exists a random variable $Z$ such that $Z_{n} \xrightarrow{n \to \infty} Z$ with probability $1$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    Note that 2.~implies 1.~since if $Z_{n} \geq 0$, then $\E[\abs{Z_{n}}] = \E[Z_{n}] = \E[Z_{1}]$ implies that $\sup_{n \geq 1} \E[\abs{Z_{n}}] < \infty$. Also, 2.~implies 3.~since $\E[\abs{Z_{n}}]^{2} \leq \E[Z_{n}^{2}]$ by Jensen's inequality. For a proof of 3., note that $\sup_{n \geq 1} \E[Z_{n}^{2}] < \infty$ implies that $\{Z_{n}^{2}\}_{n \geq 1}$ is a submartingale with bounded expectation. Taking expectation on $\E[Z_{n}^{2} \mid Z_{n-1},\ldots,Z_{1}] \geq \E[Z_{n-1}^{2}]$ gives $\E[Z_{n}^{2}] \geq \E[Z_{n-1}^{2}]$, and hence $\E[Z_{n}^{2}]$ is non-decreasing. Since it is bounded, we have $\E[Z_{n}^{2}] \xrightarrow{n \to \infty} \alpha$ for some $0 \leq \alpha < \infty$. Now fix $m \geq 1$, and set $Y_{n} = Z_{n+m}-Z_{m}$ for $n \geq 1$. Then, $\{Y_{n}\}_{n \geq 1}$ is a martingale, and Kolmogorov's maximal inequality gives
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq n} \abs{Y_{i}} > b \right) \leq \frac{\E[Y_{n}^{2}]}{b^{2}}
    \end{align}
    for all $b > 0$, and also $\E[Y_{n}^{2}] = \E[Z_{n+m}^{2}] - 2\E[Z_{n+m}Z_{m}] + \E[Z_{m}^{2}]$. Since $\{Z_{n}\}_{n \geq 1}$ is a martingale, we have $\E[Z_{n+m} \mid Z_{m},\ldots,Z_{1}] = Z_{m}$, and hence $\E[Z_{n+m}Z_{m}] = \E[\E[Z_{n+m}Z_{m} \mid Z_{m},\ldots,Z_{1}]] = \E[Z_{m}^{2}]$. Thus, $\E[Y_{n}^{2}] = \E[Z_{n+m}^{2}] - \E[Z_{m}^{2}]$, which is bounded by $\alpha - \E[Z_{m}^{2}]$ as $n \to \infty$. So,
    \begin{align}
        \Prob\left( \max_{1 \leq i \leq n} \abs{Z_{i+m}-Z_{i}} > b\right) \leq \frac{\E[Z_{n+m}^{2}]-\E[Z_{m}^{2}]}{b^{2}}
    \end{align}
    for all $n,m \geq 1$. Letting $n \to \infty$ on both sides gives
    \begin{align}
        \Prob\left( \bigcup_{i \geq 1} \{\abs{Z_{i+m}-Z_{i}} > b\} \right) \leq \frac{\alpha - \E[Z_{m}^{2}]}{b^{2}}.
    \end{align}
    Now letting $m \to \infty$ gives
    \begin{align}
        \lim_{m \to \infty} \Prob\left( \bigcup_{i \geq 1} \{\abs{Z_{i+m}-Z_{i}} > b\} \right) = 0 \qquad \text{ for all } b > 0.
    \end{align}
    We claim that this implies that $\{Z_{n}\}_{n \geq 1}$ is a Cauchy sequence with probability $1$. Indeed, look at
    \begin{align}
        A = \bigcap_{b > 0} \bigcup_{N=1}^{\infty} \bigcap_{m = N+1}^{\infty} \bigcap_{i=N+1}^{\infty} \{\abs{Z_{i+m}-Z_{i}} < b\}.
    \end{align}
    Then, $A$ is the event that $\{Z_{n}\}_{n \geq 1}$ is a Cauchy sequence; it is enough to show that $\Prob(A) = 1$. Let $l \geq 1$ and set $b = \frac{1}{2^{l}}$. By the above, there exists $k_{l}$ such that
    \begin{align}
        \Prob\left( \bigcup_{i \geq 1} \abs{Z_{i+k_{l}}-Z_{i}} > \frac{1}{2^{l}} \right) \leq \frac{1}{2^{l}}.
    \end{align}
    This implies
    \begin{align}
        \sum_{l \geq 1} \Prob\left( \bigcup_{i \geq 1} \abs{Z_{i+k_{l}}-Z_{i}} > \frac{1}{2^{l}} \right) < \infty \implies \sum_{l \geq 1} \Prob\left(  \abs{Z_{i+k_{l}}-Z_{i}} \leq \frac{1}{2^{l}} \right) < \infty.
    \end{align}
    By the Borel-Cantelli lemma, we receive
    \begin{align}
        \Prob\left( \bigcap_{n=1}^{\infty} \bigcup_{m=n}^{\infty} A_{m} \right) = 0.
    \end{align}
    Taking complements gives $\Prob(A) = 1$, and hence $\{Z_{n}\}_{n \geq 1}$ is a Cauchy sequence with probability $1$. Since $\R$ is complete, there exists a random variable $Z$ such that $Z_{n} \xrightarrow{n \to \infty} Z$ with probability $1$.
