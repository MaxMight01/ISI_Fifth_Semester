\section{Independence}
\textit{September 15th.}

\begin{definition}
    Let $(\Omega, \mathcal{F}, P)$ be a probability space.
    \begin{enumerate}
        \item We say sub-$\sigma$-algebras $\cG_{1},\cG_{2},\cdots,\cG_{n}$ of $\cF$ are independent if $\Prob(A_{1} \cap A_{2} \cap \cdots \cap A_{n}) = \Prob(A_{1})\Prob(A_{2})\ldots\Prob(A_{n})$ for all $A_{i} \in \cG_{i}$, $i=1,2,\ldots,n$.
        \item We say that the random variables $X_{1},X_{2},\ldots,X_{n}$ are independent if the sub-$\sigma$-algebras $\sigma(X_{1}),\sigma(X_{2}),\ldots,\sigma(X_{n})$ are independent.
        \item We say that the events $A_{1},A_{2},\ldots,A_{n}$ are independent if the sub-$\sigma$-algebras $\sigma(A_{1}),\sigma(A_{2}),\ldots,\sigma(A_{n})$ are independent.
    \end{enumerate}
\end{definition}

Equivalent condition for independence of events is that
\begin{align}
    \Prob(A_{1}^{\pm} \cap A_{2}^{\pm} \cap \cdots \cap A_{n}^{\pm}) = \Prob(A_{1}^{\pm})\Prob(A_{2}^{\pm})\cdots\Prob(A_{n}^{\pm}),
\end{align}
where $A_{i}^{+} = A_{i}$ and $A_{i}^{-} = A_{i}^{c}$ for $i=1,2,\ldots,n$.
\begin{definition}
    We say that a collection of countably many random variables $X_{1},X_{2},\ldots$ on $(\Omega,\cF,\Prob)$ are independent if every finite subcollection of them are independent. 
\end{definition}

\begin{definition}
    Suppose $X_{1},X_{2},\ldots$ are random variables on $(\Omega,\cF,\Prob)$, and define $\cF_{n} = \sigma(X_{n},X_{n+1},\ldots)$ for $n=1,2,\ldots$. The \eax{tail $\sigma$-algebra} of the sequence $X_{1},X_{2},\ldots$ is defined as
    \begin{align}
        \cT = \bigcap_{n=1}^{\infty} \cF_{n} \subseteq \sigma(X_{1},X_{2},\ldots) = \cF.
    \end{align}
\end{definition}

An event $A \in \cT$ is an event whose occurrence is determined by the
variables $X_{n},X_{n+1},\ldots$ for every $n$. Thus, $A$ does not depend on any fixed finite number of the variables at the beginning of the sequence. For example, the event
\begin{align}
    A = \{X_{n} \text{ occurs infinitely often}\}
\end{align}
belongs to $\cT$, since for every $m$ it can be described using only
$X_{m},X_{m+1},\ldots$.

If $X_{1},X_{2},\ldots$ are independent and $A \in \cT$, then $A$ is
independent of $\sigma(X_{1},\ldots,X_{n})$ for every $n$, since $A$ is
measurable with respect to $\sigma(X_{n+1},X_{n+2},\ldots)$. Therefore $A$ is
independent of
\begin{align}
    \sigma(X_{1},X_{2},\ldots) = \sigma\left(\bigcup_{n=1}^{\infty}
    \sigma(X_{1},\ldots,X_{n})\right).
\end{align}
In particular, $A$ is independent of itself, and hence
\begin{align}
    \Prob(A) = \Prob(A \cap A) = \Prob(A)^{2}.
\end{align}
Thus $\Prob(A) \in \{0,1\}$. This is known as \eax{Kolmogorov's zero--one law}, that for independent random variables, any tail event has probability 0 or 1. Without independence, a tail event may retain information about the variables in every finite initial segment, and this argument does not apply.

\subsection{Percolation}
\textit{September 18th.}

We first discuss (Bernoulli) percolation on $\Z^{2} \subseteq \R^{2}$. To each edge $e$, we associate $X_{e} \sim \Ber(p)$ for some $p \in [0,1]$, independently for each edge. If $X_{e} = 1$, we retain the edge $e$, and if $X_{e} = 0$, we delete the edge $e$. We then have a random subgraph of $\Z^{2}$, which is a random subset of edges of $\Z^{2}$. The present or retained edges are termed open, and the absent or deleted edges are termed closed. We ask if there exists an unbounded connected component of open edges; let $E_{p}$ be the event that there exists an unbounded connected component of open edges. Since there are only countably many $X_{e}$'s, we can order them in a spiral like manner starting from the origin, and let $X_{1},X_{2},\ldots$ be the corresponding sequence of random variables. Then $E_{p}$ is a tail event here, and hence by Kolmogorov's zero--one law, $\Prob(E_{p}) \in \{0,1\}$. Moreover, $\Prob(E_{0}) = 0$ and $\Prob(E_{1}) = 1$, and for $0 < p < q < 1$, we have $\Prob(E_{p}) \leq \Prob(E_{q})$ by a coupling argument. Therefore, there exists a critical probability $p_{c} \in [0,1]$ such that $\Prob(E_{p}) = 0$ for $p < p_{c}$ and $\Prob(E_{p}) = 1$ for $p > p_{c}$.

Let $\{0 \leftrightarrow \infty\}$ denote the event that there exists an unbounded connected component of open edges containing the origin. Define $\theta_{p} \defeq \Prob(0 \leftrightarrow \infty)$. Note that this is not a tail event, but $\theta_{p} > 0$ if and only if $\Prob(E_{p}) = 1$. Also, $\theta_{p}$ is non-decreasing in $p$.

\begin{lemma}
    For small $p$, $\Prob(E_{p}) = 0$. For $p \approx 1$, $\Prob(E_{p}) = 1$.
\end{lemma}
\begin{proof}
    Firstly, note that the event $\{0 \leftrightarrow \infty\}$ is contained in the event that there exists a self avoiding path of open edges starting from the origin and ending at manhattan distance $N$ from the origin, for some $N \in \N$. This event is further contained in the event that there exists a self avoiding path of open edges starting from the origin with length at least $N$. Call the first event $A_{N}$ and the second event $B_{N}$. Then $\{0 \leftrightarrow \infty\} \subseteq A_{N} \subseteq B_{N}$, and hence $\Prob(0 \leftrightarrow \infty) \leq \Prob(A_{N}) \leq \Prob(B_{N})$. If $C_{n}$ is the event that there exists a self avoiding path of open edges starting from the origin of length exactly $N$, then $\Prob(C_{N}) \leq 4 \cdot 3^{N-1} p^{N} \leq (4p)^{N}$, since there are at most $4 \cdot 3^{N-1}$ self avoiding paths of length $N$ starting from the origin, and each such path is open with probability $p^{N}$. Moreover, note that $B_{N} = \bigcup_{n=N}^{\infty} C_{n}$, and hence $\Prob(B_{N}) \leq \sum_{n=N}^{\infty} (4p)^{n}$. Therefore, if $p < 1/4$, we have $\Prob(B_{N}) = \frac{(4p)^{N}}{1-4p} \to 0$ as $N \to \infty$, and hence $\Prob(0 \leftrightarrow \infty) = 0$ for $p < 1/4$. So, for small $p$, specifically for $p < 1/4$, we have $\theta_{p} = 0$ and hence $\Prob(E_{p}) = 0$.

    For the second statement, note that the $\Ber(p)$ percolation model on $\Z^{2}$ induces a $\Ber(1-p)$ percolation model on the dual lattice $(\Z^{2})^{*}$. Thus, the event $\{0 \not\leftrightarrow \infty\}$ is exactly the event that there exists a loop of open edges in the dual lattice surrounding the origin. If $C_{N}^{*}$ is the event that there exists a loop of open edges in the dual lattice surrounding the origin with length exactly $N$, then this event is contained in the event that there exists a self avoiding path of open edges in the dual lattice starting at $\{(1,0),(2,0),\ldots,(n,0)\}$ of length exactly $N$. Therefore, $\Prob(C_{N}^{*}) \leq n(4(1-p))^{N}$, and hence $\Prob(\{0 \not\leftrightarrow \infty\}) \leq \sum_{N=1}^{\infty} n(4(1-p))^{N}$. Therefore, if $p > 3/4$, we have $\Prob(\{0 \not\leftrightarrow \infty\}) = 0$, and hence $\theta_{p} = 1$ and $\Prob(E_{p}) = 1$ for $p > 3/4$.
\end{proof}

\begin{lemma}[\eax{Borel-Cantelli lemma}]
    Let $A_{1},A_{2},\ldots$ be a sequence of events in a probability space $(\Omega,\cF,\Prob)$.
    \begin{enumerate}
        \item If $\sum_{n=1}^{\infty} \Prob(A_{n}) < \infty$, then $\Prob(A_{n} \text{ i.o.}) = 0$.
        \item If $A_{1},A_{2},\ldots$ are independent and $\sum_{n=1}^{\infty} \Prob(A_{n}) = \infty$, then $\Prob(A_{n} \text{ i.o.}) = 1$.
    \end{enumerate}
\end{lemma}
We prove the second statement.
\begin{proof}
    Let $B_{n} = \bigcup_{j=n}^{\infty} A_{j}$, so that $B_{1} \supseteq B_{2} \supseteq \cdots$ and $\Prob(A_{n} \text{ i.o.}) = \Prob(\bigcap_{n=1}^{\infty} B_{n})$. If we prove that $\Prob(B_{n}) = 1$ for all $n$, then we are done. Now, $\Prob(B_{n}) = 1 - \Prob(B_{n}^{c}) = 1 - \Prob(\bigcap_{j=n}^{\infty} A_{j}^{c})$. Since $A_{1},A_{2},\ldots$ are independent, we have
    \begin{align}
        \Prob(\bigcap_{j=n}^{m} A_{j}^{c}) = \prod_{j=n}^{m} \Prob(A_{j}^{c}) = \prod_{j=n}^{m} (1 - \Prob(A_{j})) \leq \prod_{j=n}^{m} e^{-\Prob(A_{j})} = e^{-\sum_{j=n}^{m} \Prob(A_{j})}.
    \end{align}
    Taking the limit as $m \to \infty$, we get $\Prob(\bigcap_{j=n}^{\infty} A_{j}^{c}) \leq e^{-\sum_{j=n}^{\infty} \Prob(A_{j})} = 0$, since $\sum_{j=n}^{\infty} \Prob(A_{j}) = \infty$. Therefore, $\Prob(B_{n}) = 1$ for all $n$, and hence $\Prob(A_{n} \text{ i.o.}) = 1$.
\end{proof}

\section{Moments}
For a random variable $X:(\Omega,\cF,\Prob) \to \R$, recall that the \eax{mean} of $X$ was simply its expectation $\E[X]$, if it exists. Else, the mean is undefined. For higher moments, the \eax{$p$-th moment} of $X$ is defined as $\E[\abs{X}^{p}]$, if it exists. The \eax{variance} of $X$ is defined as $\Var(X) = \E[(X - \E[X])^{2}] = \E[X^{2}] - \E[X]^{2}$.

In many cases, for a given $X$, we may not know its exact distribution, but we may know its moments. In such cases, we can use the moments to bound the probabilities of certain events. For example, if $X$ is a non-negative random variable with finite mean $\E[X]$, then by Markov's inequality, for any $t > 0$, we have
\begin{align}
    \Prob(X > t) \leq \frac{\E[X]}{t}.
\end{align}
In particular, if $\E[X] = 0$, then $\Prob(X \geq t) = 0$ for all $t > 0$, and hence $X = 0$ almost surely. If $X$ also happens to have a finite second moment $\E[X^{2}]$, then we can replace $X$ with $\abs{X-\E[X]}$ in the above inequality to get Chebyshev's inequality, which states that for any $t > 0$,
\begin{align}
    \Prob(\abs{X - \E[X]} > t) \leq \frac{\Var(X)}{t^{2}}.
\end{align}
In fact, if $\E[\abs{X-\E[X]}^{n}] < \infty$, then
\begin{align}
    \Prob(\abs{X-\mu} > t) \leq \frac{\E[\abs{X-\mu}^{n}]}{t^{n}}.
\end{align}
The \eax{exponential moment} of $X$ is defined as $\E[e^{\lambda x}]$ for $\lambda \neq 0$, if it exists. If $\E[e^{\lambda x}] < \infty$ for some $\lambda > 0$, then by Markov's inequality, we have
\begin{align}
    \Prob(X > t) = \Prob(e^{\lambda X} > e^{\lambda t}) \leq \frac{\E[e^{\lambda X}]}{e^{\lambda t}}.
\end{align}
Note that if the $p$-th moment of $X$ exists, then the $q$-th moment of $X$ also exists for all $0 < q < p$; this holds since if we write $\abs{X}^{p} = \abs{X}^{p}\1_{\{\abs{X} \leq 1\}} + \abs{X}^{p}\1_{\{\abs{X} > 1\}}$, and note that $\E[\abs{X}^{p}]$ is finite if and only if $\E[\abs{X}^{p}\1_{\{\abs{X} > 1\}}]$ is finite, then
\begin{align}
    \E[\abs{X}^{q}] = \E[\abs{X}^{q}\1_{\{\abs{X} \leq 1\}}] + \E[\abs{X}^{q}\1_{\{\abs{X} > 1\}}] \leq 1 + \E[\abs{X}^{p}\1_{\{\abs{X} > 1\}}] < \infty.
\end{align}

\section{Convergence of Random Variables}
\textit{September 22nd.}

Throughout, we shall assume that $(X_{n})_{n \geq 1}$ is a sequence of random variables on $(\Omega,\cF,\Prob)$, and $X$ is another random variable on the same probability space. We shall discuss various notions of convergence of the sequence $(X_{n})$ to $X$.

\begin{definition}
    $X_{n}$ converges to $X$ as \eax{almost sure convergence}, written $X_{n} \toas X$, if there exists a set $E \in \cF$ with $\Prob(E) = 0$ such that
    \begin{align}
        \lim_{n \to \infty} X_{n}(\omega) = X(\omega) \quad \text{for all } \omega \in \Omega \setminus E.
    \end{align}
\end{definition}

\begin{definition}
    $X_{n}$ converges to $X$ as \eax{convergence in probability}, written $X_{n} \toP X$, if
    \begin{align}
        \lim_{n \to \infty} \Prob(\abs{X_{n}-X} > \epsilon) = 0 \quad \text{for all } \epsilon > 0.
    \end{align}
\end{definition}
Almost sure convergence is the stronger notion---
\begin{proposition}
    If $X_{n} \toas X$, then $X_{n} \toP X$.
\end{proposition}
\begin{proof}
    Let $\epsilon > 0$. Since $X_{n} \toas X$, there exists a set $E \in \cF$ with $\Prob(E) = 0$ such that $\lim_{n \to \infty} X_{n}(\omega) = X(\omega)$ for all $\omega \in \Omega \setminus E$. Therefore, for each $\omega \in \Omega \setminus E$, there exists $N_{\epsilon}(\omega) \in \N$ such that $\abs{X_{n}(\omega)-X(\omega)} \leq \epsilon$ for all $n \geq N_{\epsilon}(\omega)$. Now define
    \begin{align}
        A_{k} \defeq \{\omega \in \Omega \mid \abs{X_{n}(\omega)-X(\omega)} \leq \epsilon, \; \text{for all } n \geq k\}.
    \end{align}
    Then $(A_{k})_{k \geq 1}$ is an increasing sequence of sets such that $\emptyset \subseteq A_{k} \uparrow \Omega\setminus E$; for any $(1 > ) \delta > 0$, there exists a natural $N \in \N$ such that $\Prob(A_{N}) \geq 1-\delta$. Then, we must have
    \begin{align}
        \{\abs{X_{n}-X} > \epsilon\} \subseteq \Omega \setminus A_{N} \implies 0 \leq \Prob(\abs{X_{n}-X} > \epsilon) \leq \Prob(\Omega \setminus A_{N}) \leq \delta.
    \end{align}
    Letting $\delta \to 0^{+}$, we get $\Prob(\abs{X_{n}-X} > \epsilon) \to 0$ as $n \to \infty$; since $\epsilon > 0$ was arbitrary, we have $X_{n} \toP X$.
\end{proof}

However, the converse does not hold. Consider the sequence of random variables $(X_{n})_{n \geq 1}$ on $([0,1],\cB([0,1]),\lambda)$ defined as
\begin{align}
    X_{2^{k}+j}(x) &= \1_{[j2^{-k},(j+1)2^{-k})}(x),
    \qquad k=0,1,2,\ldots,\quad j=0,1,\ldots,2^{k}-1.
\end{align}
Let $X=0$. For $n=2^{k}+j$, we have
\begin{align}
    \Prob(\abs{X_n-X}>\epsilon) = 2^{-k} \longrightarrow 0
    \qquad (0<\epsilon<1),
\end{align}
and the probability is zero for $\epsilon\geq 1$. Hence $X_n\toP 0$. On the other hand, every $x\in[0,1)$ belongs to exactly one interval at each level $k$, so $X_n(x)=1$ infinitely often, while $X_n(x)=0$ infinitely often. Thus $X_n$ does not converge almost surely to $0$ (indeed, it fails to converge for every $x\in[0,1)$).

\begin{lemma}
    Suppose $X_{n} \toP X$ and, additionally,
    \begin{align}
        \sum_{n=1}^{\infty} \Prob(\abs{X_{n}-X} > \epsilon) < \infty
        \qquad \text{for all } \epsilon > 0.
    \end{align}
    Then $X_{n} \toas X$.
\end{lemma}
\begin{proof}
    Let $\epsilon > 0$. By the Borel-Cantelli lemma, we have $\Prob(\abs{X_{n}-X} > \epsilon \text{ i.o.}) = 0$. Let $E_{\epsilon} = \{\abs{X_{n}-X} > \epsilon \text{ i.o.}\}$, so that $\Prob(E_{\epsilon}) = 0$. Then for all $\omega \in \Omega \setminus E_{\epsilon}$, there exists $N_{\epsilon}(\omega) \in \N$ such that $\abs{X_{n}(\omega)-X(\omega)} \leq \epsilon$ for all $n \geq N_{\epsilon}(\omega)$. Now define
    \begin{align}
        E = \bigcup_{m=1}^{\infty} E_{1/m}.
    \end{align}
    Then $\Prob(E) = 0$, and for all $\omega \in \Omega \setminus E$, we have $\abs{X_{n}(\omega)-X(\omega)} \leq 1/m$ for all $n$ sufficiently large, for every $m=1,2,\ldots$. Therefore, $\lim_{n\to\infty} X_{n}(\omega) = X(\omega)$ for all $\omega \in \Omega\setminus E$, and hence $X_{n} \toas X$.
\end{proof}

Let $(X_{n})_{n \geq 1}$ be a sequence of pairwise independent random variables with $\E[X_{n}] = 0$ and $\Var(X_{n}) \leq M$ for all $n$ and some constant $M > 0$. Then if $S_{n} = X_{1} + \cdots + X_{n}$, we have $S_{n}/n \toP 0$ since
\begin{align}
    \Prob(\abs{S_{n}/n} > \epsilon) \leq \frac{\E[\abs{\frac{S_{n}}{n}}^{2}]}{\epsilon^{2}} = \frac{\Var(S_{n}/n)}{\epsilon^{2}} = \frac{\Var(S_{n})}{n^{2}\epsilon^{2}} = \frac{\sum_{i=1}^{n} \Var(X_{i})}{n^{2}\epsilon^{2}} \leq \frac{M}{n\epsilon^{2}} \to 0 \quad \text{as } n \to \infty.
\end{align}
Let $(X_{n})_{n \geq 1}$ be bounded independent random variables with $\E[X_{n}] = 0$ for all $n$. Then if $S_{n} = X_{1} + \cdots + X_{n}$, we have $S_{n}/n \toas 0$; to this end, we have
\begin{align}
    \Prob(\abs{S_{n}/n} > \epsilon) \leq \frac{\E[\abs{S_{n}/n}^{4}]}{\epsilon^{4}} = \frac{\E[S_{n}^{4}]}{n^{4}\epsilon^{4}} \leq \frac{C(n+n^{2})M^{4}}{n^{4}\epsilon^{4}} \lesssim \frac{1}{n^{2}}.
\end{align}
Thus,
\begin{align}
    \sum_{n=1}^{\infty} \Prob(\abs{S_{n}/n} > \epsilon) < \infty,
\end{align}
or $\abs{S_{n}/n} \toas 0$ by the previous lemma.

\section{Second Moment Method}
\textit{September 25th.}

Let $X \geq 0$ be a non-negative random variable. Given just this, note that $\Prob(X > 0)$ cannot have a lower bound since we could very well define $X$ to be $0$ with probability $1-\epsilon$ and $1$ with probability $\epsilon$, for any $\epsilon > 0$. If we specify that $\E[X]$ has to be `large', we stil cannot guarantee that $\Prob(X > 0)$ is large, since we could define $X$ to be $0$ with probability $1-\epsilon$ and $1/\epsilon$ with probability $\epsilon$, for any $1 > \epsilon > 0$.

Suppose we are give $\mu = \E[X]$. Then by Markov's inequality, we have
\begin{align}
    \Prob(X > \frac{\mu}{2}) = \Prob(\abs{X-\mu} \leq \frac{\mu}{2}) + \Prob(X > \frac{3 \mu}{2}) \geq 1 - \Prob(\abs{X-\mu} > \frac{\mu}{2}) \geq 1 - \frac{4\Var(X)}{\mu^{2}} \geq 1-4\delta
\end{align}
when $\Var(X) \leq \delta \mu^{2}$ for some $\delta > 0$. Thus, if $\Var(X)$ is small compared to $\E[X]^{2}$, then $X$ is concentrated around its mean, and hence $\Prob(X > 0)$ is large. Note that the choice of $\mu/2$ is arbitrary; we have the following generalization.
\begin{lemma}[The \eax{second moment method}]
    Let $X \geq 0$ be a non-negative random variable, and $0 \leq \alpha \leq 1$. Then
    \begin{align}
        \Prob(X > \alpha \E[X]) \geq (1-\alpha)^{2} \frac{\E[X]^{2}}{\E[X^{2}]}.
    \end{align}
    In particular,
    \begin{align}
        \Prob(X > 0) \geq \frac{\E[X]^{2}}{\E[X^{2}]}.
    \end{align}
\end{lemma}
\begin{proof}
    Recall the Cauchy-Schwarz inequality, which states that for any random variables $X$ and $Y$, we have $\E[XY]^{2} \leq \E[X^{2}]\E[Y^{2}]$. Let $Y = \1_{\{X > \alpha \E[X]\}}$. Then $\E[Y^{2}] = \E[Y] = \Prob(X > \alpha \E[X])$, and $\E[XY] = \E[X\1_{\{X > \alpha \E[X]\}}] = \E[X] - \E[X\1_{\{X \leq \alpha \E[X]\}}] \geq (1-\alpha)\E[X]$. Therefore, by Cauchy-Schwarz,
    \begin{align}
        \Prob(X > \alpha \E[X]) = \E[Y^{2}] \geq \frac{\E[XY]^{2}}{\E[X^{2}]} \geq (1-\alpha)^{2} \frac{\E[X]^{2}}{\E[X^{2}]}.
    \end{align}
\end{proof}

We may also prove the Borel-Cantelli lemma using the second moment method, the statement being that if $A_{1},A_{2},\ldots$ are independent events (or simply pairwise independent) with $\sum_{n=1}^{\infty} \Prob(A_{n}) = \infty$, then $\Prob(A_{n} \text{ i.o.}) = 1$. 
\begin{proof}
    Let $B_{n} = \bigcup_{i=n}^{\infty} A_{i}$. Then $B_{n} \downarrow \{A_{n} \text{ i.o.}\}$. Fix $n$ and define $S_{k} \defeq \sum_{j=n}^{k} \1_{A_{j}}$ for $k \geq n$. Then, with $\Prob(A_{i}) = \alpha_{i}$,
    \begin{align}
        \Prob(S_{k} \geq 1) = \Prob(S_{k} > 0) \geq \frac{\E[S_{k}]^{2}}{\E[S_{k}^{2}]} &= \frac{(\sum_{j=n}^{k} \Prob(A_{j}))^{2}}{\sum_{j=n}^{k} \Prob(A_{j}) + 2\sum_{n \leq i < j \leq k} \Prob(A_{i} \cap A_{j})} \notag \\ &= \frac{(\sum_{j=n}^{k} \alpha_{j})^{2}}{\sum_{j=n}^{k} \alpha_{j} + 2\sum_{n \leq i < j \leq k} \alpha_{i} \alpha_{j}} = \frac{(\sum_{i=n}^{k} \alpha_{i})^{2}}{(\sum_{i=n}^{k} \alpha_{i})+(\sum_{i=n}^{k} \alpha_{i})^{2}} \notag \\ &\geq \dfrac{1}{1+\frac{1}{\sum_{i=n}^{k} \alpha_{i}}}.
    \end{align}
    Letting $k \to \infty$, we get
    \begin{align}
        \Prob(B_{n}) = \lim_{k \to \infty} \Prob(S_{k} \geq 1) \geq 1.
    \end{align}
    Since this holds for all $n$, we have $\Prob(A_{n} \text{ i.o.}) = 1$.
\end{proof}

\subsection{The Coupon Collector Problem}
$n$ different books are placed on a bookshelf. Every night, a book is chosen uniformly at random, read, and then returned to the shelf. Let $T_{n}$ be the number of days until all $n$ books have been read. We want to find $\E[T_{n}]$. To this end, define $Z_{1} = 1$, and for $2 \leq k \leq n$, let $Z_{k}$ be the number of days after the first $k-1$ books have been read until a new book is read. Then $T_{n} = Z_{1} + Z_{2} + \cdots + Z_{n}$. Note that $Z_{2} \sim \Geom(\frac{n-1}{n})$, since there are $n-1$ books that have not been read yet with probability $\frac{n-1}{n}$ of reading a new book, and similarly $Z_{k} \sim \Geom(\frac{n-k+1}{n})$ for $2 \leq k \leq n$. Therefore, we have
\begin{align}
    \E[T_{n}] = \sum_{k=1}^{n} \E[Z_{k}] = \sum_{k=1}^{n} \frac{n}{n-k+1} = n\sum_{k=1}^{n} \frac{1}{k} \sim n\log n.
\end{align}
Now let $\theta_{n}$ be a sequence of positive numbers such that $\theta_{n} \to \infty$ as $n \to \infty$. We claim that
\begin{align}
    \Prob(\abs{T_{n}-n \log n} \geq n \theta_{n}) \to 0 \quad \text{as } n \to \infty.
\end{align}
To show this, set $t \in \N$, and define
\begin{align}
    S_{t} \defeq \sum_{j=1}^{n} \1_{B_{j}},
\end{align}
where $B_{j}$ is the event that book $j$ has not been picked in the first $t$ days. Then $S_{t}$ is the number of books that have not been picked in the first $t$ days, and hence $\{T_{n} > t\} = \{S_{t} \geq 1\} = \{S_{t} > 0\}$. Computing moments of $S_{t}$, while later setting $t = n \log n + n \theta_{n}$, we have
\begin{align}
    \E[S_{t}] = \sum_{j=1}^{n} \Prob(B_{j}) = n\left(1-\frac{1}{n}\right)^{t} \geq ne^{-(\frac{1}{n}+\frac{1}{n^{2}})t} = e^{\theta_{n} - \frac{1}{n}(\log n -\theta_{n})}.
\end{align}
and
\begin{align}
    \E[S_{t}^{2}] = \sum_{j=1}^{n} \Prob(B_{j}) + 2\sum_{1 \leq i < j \leq n} \Prob(B_{i} \cap B_{j}) &= n\left(1-\frac{1}{n}\right)^{t} + n(n-1)\left(1-\frac{2}{n}\right)^{t} \notag \\
    &\leq ne^{-\frac{t}{n}} + n^{2}e^{-\frac{2t}{n}} = e^{\theta_{n}}+e^{2\theta_{n}}.
\end{align}
Therefore, by the second moment method, we have
\begin{align}
    \Prob(T_{n} > n \log n - n \theta_{n}) = \Prob(S_{t} > 0) \geq \frac{\E[S_{t}]^{2}}{\E[S_{t}^{2}]} \geq \frac{e^{2\theta_{n}-\frac{2}{n}(\log n -\theta_{n})}}{e^{\theta_{n}}+e^{2\theta_{n}}} \implies \Prob(T_{n} > n \log n - n \theta_{n}) \xrightarrow{n \to \infty} 1.
\end{align}
On the other hand, setting $t = n \log n + n \theta_{n}$ and using Markov's inequality, we have
\begin{align}
    \Prob(T_{n} > n \log n + n \theta_{n}) = \Prob(S_{t} \geq 1) \leq \E[S_{t}] \leq n\left(1-\frac{1}{n}\right)^{t} &\leq ne^{-\frac{1}{n} n(\log n + \theta_{n})} = e^{-\theta_{n}} \notag \\ &\implies \Prob(T_{n} > n \log n + n \theta_{n}) \xrightarrow{n \to \infty} 0.
\end{align}
Therefore, we conclude that
\begin{align}
    \Prob(\abs{T_{n}-n \log n} \geq n \theta_{n}) = \Prob(T_{n} > n \log n + n \theta_{n}) + \Prob(T_{n} < n \log n - n \theta_{n}) \xrightarrow{n \to \infty} 0.
\end{align}
Dividing by $n \log n$, and setting $\theta_{n} = \epsilon \log n$ for any $\epsilon > 0$, we get
\begin{align}
    \frac{T_{n}}{n \log n} \toP 1.
\end{align}

\subsection{Random Series}
Let $(a_{n})_{n \geq 0}$ be a sequence of independent and identically distribution random variables with $a_{n} \sim \Exp(1)$, which has a density function $e^{-x}\1_{x \geq 0}$. Now define
\begin{align}
    f(t) = \sum_{n=0}^{\infty} a_{n} t^{n} \qquad \text{for all } t \in \R.
\end{align}
This is a random power series, and we want to find the radius of convergence, $R$, of this series. We know that $R$ is related to the coefficients $a_{n}$ as
\begin{align}
    \frac{1}{R} = \limsup_{n \to \infty} \abs{a_{n}}^{1/n}.
\end{align}
This limit is a random variable which is measurable with respect to the tail-$\sigma$-algebra of the sequence $(a_{n})_{n \geq 0}$, and hence by Kolmogorov's zero-one law, there exists a constant $c \in [0,\infty]$ such that $\Prob(R = c) = 1$. We claim that $c = 1$. First note that if we set $t = 1+\epsilon$ for some $\epsilon > 0$, then
\begin{align}
    \Prob(\abs{a_{n}} > t^{n}) = e^{-t^{n}} \leq \frac{1}{(1+\epsilon)^{n}} \quad \text{for all } n \text{ sufficiently large}.
\end{align}
Thus, $\sum_{n=0}^{\infty} \Prob(\abs{a_{n}} > t^{n}) < \infty$, and hence by the Borel-Cantelli lemma, we have $\Prob(\abs{a_{n}}^{1/n} > t \text{ i.o.}) = 0$, or $\Prob(\limsup_{n \to \infty} \abs{a_{n}}^{1/n} > t) = 0$. Since this holds for all $t > 1$, we have $\Prob(\limsup_{n \to \infty} \abs{a_{n}}^{1/n} > 1) = 0$. On the other hand, if we set $t = 1-\epsilon$ for some $\epsilon > 0$, then 
\begin{align}
    \sum_{n=0}^{\infty} \Prob(\abs{a_{n}} > t^{n}) = \sum_{n=0}^{\infty} e^{-t^{n}} = \infty,
\end{align}
so that by the Borel-Cantelli lemma, we have $\Prob(\abs{a_{n}}^{1/n} > t \text{ i.o.}) = 1$, or $\Prob(\limsup_{n \to \infty} \abs{a_{n}}^{1/n} > t) = 1$. Since this holds for all $t < 1$, we have $\Prob(\limsup_{n \to \infty} \abs{a_{n}}^{1/n} < 1) = 0$. Therefore, we conclude that $\Prob(\limsup_{n \to \infty} \abs{a_{n}}^{1/n} = 1) = 1$, and hence $\Prob(R = 1) = 1$.

\section{Law of Large Numbers}

\textit{September 29th.}

\begin{theorem}[The \eax{basic weak law of large numbers}]
    Let $(X_{n})_{n \geq 1}$ be a sequence of independent and identically distributed random variables with $\E[\abs{X_{1}}] < \infty$ and $\Var(X_{1}) < \infty$. Then
    \begin{align}
        \frac{S_{n}}{n} =\frac{X_{1} + \cdots + X_{n}}{n} \toP \mu = \E[X_{1}].
    \end{align}
\end{theorem}
\begin{proof}
    We wish to show that for all $\epsilon > 0$, we have $\Prob(\abs{S_{n}/n - \mu} > \epsilon) \to 0$ as $n \to \infty$. To this end, we first note that by Chebyshev's inequality, we have
    \begin{align}
        \Prob\left( \abs{\frac{S_{n}}{n}-\mu} > \epsilon \right) = \Prob\left( \abs{\sum_{i=1}^{n}(X_{i}-\mu)} > n\epsilon \right) \leq \frac{\E\left[ \left( \sum_{i=1}^{n}(X_{i}-\mu) \right)^2 \right]}{n^2\epsilon^2} = \frac{n \Var(X_{1})}{n^2\epsilon^2} = \frac{\Var(X_{1})}{n\epsilon^2} \xrightarrow{n \to \infty} 0.
    \end{align}
\end{proof}

We can use the weak law of large numbers to prove the Weierstrass approximation theorem, which states that any continuous function on $[0,1]$ can be uniformly approximated by polynomials. To this end, we make use of the \eax{Bernstein polynomials}, which are defined as follows. For a continuous function $f:[0,1] \to \R$, define the $n$-th Bernstein polynomial of $f$ as
\begin{align}
    Q_{n,f}:[0,1] \to \R, \qquad Q_{n,f}(p) = \sum_{k=0}^{n} f\left(\frac{k}{n}\right) \binom{n}{k} p^{k}(1-p)^{n-k}.
\end{align}
Note that if $S_{n} \sim \Bin(n,p)$, then $\Prob(S_{n}=k) = \binom{n}{k} p^{k}(1-p)^{n-k}$, and hence $Q_{n,f}(p) = \E[f(S_{n}/n)]$. Let $S_{n} = X_{1}+X_{2}+\cdots+X_{n}$, where $X_{1},\ldots,X_{n} \sim \Ber(p)$ are independent. Then, we have
\begin{align}
    \abs{Q_{n,f}(p)-f(p)} \leq \E\left[ \abs{f\left(\frac{S_{n}}{n}\right) - f(p)} \right].
\end{align}
Splitting via the indicator function of $\{\abs{\frac{S_{n}}{n}-p} > \delta \}$ for some $\delta > 0$ we choose later, we have
\begin{align}
    \E\left[ \abs{f\left(\frac{S_{n}}{n}\right) - f(p)} \right] = \E\left[ \abs{f\left(\frac{S_{n}}{n}\right) - f(p)} \cdot \1_{\abs{\frac{S_{n}}{n}-p} \leq \delta } \right] + \E\left[ \abs{f\left(\frac{S_{n}}{n}\right) - f(p)} \cdot \1_{\abs{\frac{S_{n}}{n}-p} > \delta} \right].
\end{align}
By the uniform continuity of $f$, given $\epsilon > 0$, there exists $\delta > 0$ such that $\abs{f(x)-f(y)} < \epsilon/2$ whenever $\abs{x-y} \leq \delta$. Thus, the first of these terms may simply be bounded by $\epsilon/2$. For the second term, we have
\begin{align}
    \E\left[ \abs{f\left(\frac{S_{n}}{n}\right) - f(p)} \cdot \1_{\abs{\frac{S_{n}}{n}-p} > \delta} \right] &\leq 2\norm{f}_{\infty} \Prob\left(\abs{\frac{S_{n}}{n}-p} > \delta\right) \leq 2\norm{f}_{\infty} \frac{\Var(S_{n}/n)}{\delta^{2}} = 2\norm{f}_{\infty} \frac{p(1-p)}{n\delta^{2}}.
\end{align}
With $c_{f} = 2\norm{f}_{\infty}(1/4)$, the second term is bounded by $c_{f}/(n\delta^{2})$, which goes to $0$ as $n \to \infty$. Therefore, we have $\abs{Q_{n,f}(p)-f(p)} < \epsilon$ for all $n$ sufficiently large, and hence $Q_{n,f} \to f$ uniformly on $[0,1]$.

We now prove the weak law of large numbers, where we drop the assumption of finite variance.
\begin{theorem}[The \eax{weak law of large numbers}]
    Let $(X_{n})_{n \geq 1}$ be a sequence of independent and identically distributed random variables with $\E[\abs{X_{1}}] < \infty$. Then
    \begin{align}
        \frac{S_{n}}{n} =\frac{X_{1} + \cdots + X_{n}}{n} \toP \mu = \E[X_{1}].
    \end{align}
\end{theorem}
\begin{proof}
    Note that it suffices to prove the result for $\mu = 0$, since we can always replace $X_{n}$ with $X_{n}-\mu$. Since we have the basic weak law of large numbers for random variables with finite variance, we will approximate $X_{n}$ with a truncated version of itself to reduce to the finite variance case. To this end, for each $n \geq 1$, we can split $X_{j}$ into two parts as
    \begin{align}
        X_{j} = X_{j} \1_{\abs{X_{j}} \leq A_{n}} + X_{j} \1_{\abs{X_{j}} > A_{n}} = Y_{j,n} + Z_{j,n},
    \end{align}
    where $(A_{n})_{n \geq 1}$ is a sequence of positive numbers the diverge properly to infinity, that we choose later. The first of these terms converges pointwise to $X_{j}$ as $n \to \infty$, and is bounded uniformly; by the dominated convergence theorem, we have $\E[X_{j} \1_{\abs{X_{j}} \leq A_{n}}] \to \E[X_{j}] = 0$ as $n \to \infty$. The second of these terms converges pointwise to $0$ as $n \to \infty$, and is bounded by $\abs{X_{j}}$; again by the dominated convergence theorem, we have $\E[X_{j} \1_{\abs{X_{j}} > A_{n}}] \to 0$ as $n \to \infty$. Let $\mu_{n} = \E[Y_{1,n}] = -\E[Z_{1,n}]$. Then $\mu_{n} \to 0$ as $n \to \infty$. Now, each $S_{n}$ may be split as $S_{n} = S_{n}^{Y} + S_{n}^{Z}$, where $S_{n}^{Y} = Y_{1,n} + \cdots + Y_{n,n}$ and $S_{n}^{Z} = Z_{1,n} + \cdots + Z_{n,n}$. Then, for any $\epsilon > 0$, we have
    \begin{align}
        \Prob\left(\abs{\frac{S_{n}}{n}} > \delta \right) = \Prob\left(\abs{\frac{S_{n}^{Y}}{n} + \frac{S_{n}^{Z}}{n}} > \delta \right) \leq \Prob\left(\abs{\frac{S_{n}^{Y}}{n} - \mu_{n}} > \frac{\delta}{2} \right) + \Prob\left(\abs{\frac{S_{n}^{Z}}{n} + \mu_{n}} > \frac{\delta}{2} \right).
    \end{align}
    Via Chebyshev's, the first of these terms is bounded as
    \begin{align}
        \Prob\left(\abs{\frac{S_{n}^{Y}}{n} - \mu_{n}} > \frac{\delta}{2} \right) \leq \frac{4\Var(Y_{1,n})}{n\delta^{2}}
    \end{align}
    For $\delta > 0$ fixed, let us estimate this for large enough $n$ such that $\abs{\mu_{n}} < \frac{\delta}{4}$. For the second term, we have
    \begin{align}
        \Prob\left(\abs{\frac{S_{n}^{Z}}{n} + \mu_{n}} > \frac{\delta}{2} \right) &\leq \Prob\left(\abs{\frac{S_{n}^{Z}}{n}} > \frac{\delta}{4} \right) \leq \Prob(\text{at least one of the } Z_{j,n} \text{ is non-zero}) \leq n \Prob(\abs{X_{1}} > A_{n}).
    \end{align}
    Combining, we obtain
    \begin{align}
        \Prob\left(\abs{\frac{S_{n}}{n}} > \delta \right) \leq \frac{4\Var(Y_{1,n})}{n\delta^{2}} + n \Prob(\abs{X_{1}} > A_{n}) \leq \frac{4 A_{n} \E[\abs{X_{1}}]}{n\delta^{2}} + \frac{n\E[\abs{X_{1}} \cdot \1_{\abs{X_{1}} > A_{n}}]}{A_{n}}.
    \end{align}
    We now wish to choose $A_{n}$ such that both of these terms go to $0$ as $n \to \infty$. So, given $\epsilon > 0$, choosen $A_{n} = \epsilon n \delta^{2}/4\E[\abs{X_{1}}]$. Then, we have
    \begin{align}
        \Prob\left(\abs{\frac{S_{n}}{n}} > \delta \right) \leq \epsilon + \frac{4\E[\abs{X_{1}}]}{\delta^{2}} \epsilon \E[\abs{X_{1}} \cdot \1_{\abs{X_{1}} > A_{n}}] \xrightarrow{n \to \infty} 2\epsilon.
    \end{align}
    Since $\epsilon > 0$ was arbitrary, we conclude that $\Prob(\abs{S_{n}/n} > \delta) \to 0$ as $n \to \infty$, and hence $S_{n}/n \toP 0$.
\end{proof}

\begin{example}
    Let $X_{1},X_{2},\ldots \sim \Exp(1)$ be independent and identically distributed random variables, and set $M_{n} \defeq \max\{X_{1},\ldots,X_{n}\}$. Then $M_{n}/\log n \toP 1$ as $n \to \infty$. To see this, note that
    \begin{align}
        \Prob(M_{n} > (1+\epsilon)\log n) = 1 - \Prob(X_{1} \leq (1+\epsilon)\log n)^{n} = 1 - (1-n^{-(1+\epsilon)})^{n} \to 0,
    \end{align}
    as $n \to \infty$.
\end{example}


\section{$L^{p}$ Spaces}
\textit{October 6th.}

Fix a probability space $(\Omega,\cF,\Prob)$. For $0 < p <\infty$, define the set
\begin{align}
    V_{p} = \{X:\Omega \to \R \mid \E[\abs{X}^{p}] < \infty\}
\end{align}
and (for $p = \infty$) define
\begin{align}
    V_{\infty} = \{X:\Omega \to \R \mid \exists M > 0 \text{ such that } \Prob(\abs{X} > M) = 0\}.
\end{align}

\begin{proposition}
    $V_{p}$ and $V_{\infty}$ are real vector spaces with respect to the obvious operations of addition and scalar multiplication.
\end{proposition}
\begin{proof}
    For $p \geq 1$, let $X,Y \in V_{p}$ and $\alpha \in \R$. Then, by the triangle inequality and convexity of $x \mapsto x^{p}$, we have
    \begin{align}
        \E[\abs{X+Y}^{p}] \leq 2^{p-1}(\E[\abs{X}^{p}] + \E[\abs{Y}^{p}]) < \infty,
    \end{align}
    and
    \begin{align}
        \E[\abs{\alpha X}^{p}] = \abs{\alpha}^{p}\E[\abs{X}^{p}] < \infty.
    \end{align}
    Hence $V_{p}$ is closed under addition and scalar multiplication. For $p < 1$, we use the fact that $\abs{x+y}^{p} \leq \abs{x}^{p} + \abs{y}^{p}$, and the fact that $\abs{\alpha x}^{p} = \abs{\alpha}^{p}\abs{x}^{p}$. The proof for $V_{\infty}$ is similar.
\end{proof}

We now wish to define a norm on $V_{p}$ and $V_{\infty}$; the obvious candidate is to define $\norm{X}_{p} = \E[\abs{X}^{p}]^{1/p}$ for $0 < p < \infty$, and $\norm{X}_{\infty} = \inf\{M > 0 \mid \Prob(\abs{X} > M) = 0\}$. However, this is not a norm since $\norm{X}_{p} = 0$ implies $X = 0$ only almost surely and not exactly. The definiteness property fails. Hence, to rectify this, we define an equivalence relation on $V_{p}$ and $V_{\infty}$ as follows. For $X,Y \in V_{p}$, we say that $X \sim Y$ if $\Prob(X \neq Y) = 0$, and similarly for $X,Y \in V_{\infty}$. Then, we define the quotient spaces
\begin{align}
    L^{p} = V_{p}/\sim \qquad \text{and} \qquad L^{\infty} = V_{\infty}/\sim.
\end{align}
The norms $\norm{\cdot}_{p}$ and $\norm{\cdot}_{\infty}$ are now definite on $L^{p}$ and $L^{\infty}$, respectively. For $p < 1$, the triangle inequality still fails. Hence, $L_{\infty}$ and $L_{p}$ for $p \geq 1$ are normed vector spaces. In fact, these are Banach spaces via the Riesz-Fischer theorem, which states that $L^{p}$ and $L^{\infty}$ are complete with respect to their respective norms. We will assume this theorem without proof. The $L^{2}$ space is particularly important, since it is a Hilbert space with respect to the inner product $\ip{X,Y} = \E[XY]$.

\begin{proposition}
    Let $1 \leq p \leq q \leq \infty$. Then $L^{q} \subseteq L^{p}$.
\end{proposition}
\begin{proof}
    If $q = \infty$, then $\abs{X} \leq \norm{X}_{\infty}$ almost surely,
    and hence $X \in L^{p}$. Now suppose that $q < \infty$ and let
    $X \in L^{q}$. If $p=q$, the result is immediate. For $p<q$, we have
    \begin{align}
        \E[\abs{X}^{p}] = \E[\abs{X}^{p}\1_{\abs{X} \leq 1}] + \E[\abs{X}^{p}\1_{\abs{X} > 1}] \leq 1 + \E[\abs{X}^{q}\1_{\abs{X} > 1}] \leq 1 + \E[\abs{X}^{q}] < \infty,
    \end{align}
    and hence $X \in L^{p}$.
\end{proof}

To show triangle inequality for $L^{p}$, $p \geq 1$, we require convex functions.
\begin{definition}
    A function $\varphi:(a,b) \to \R$ is a \eax{convex function} if $\varphi$ is continuous and given any $u \in (a,b)$, there exists a linear function $L_{u}(x) = \alpha_{u}x + \beta_{u}$ such that $L_{u}(u) = \varphi(u)$ and $L_{u}(x) \leq \varphi(x)$ for all $x \in (a,b)$. In other words, there exists a line passing through $(u,\varphi(u))$ that lies below the graph of $\varphi$.
\end{definition}

\begin{lemma}[\eax{Jensen's inequality}]
    Let $\varphi:\R \to \R$ be a convex function, and suppose $X,\varphi(X) \in L^{1}$. Then $\varphi(\E[X]) \leq \E[\varphi(X)]$.
\end{lemma}
\begin{proof}
    Set $a = \E[X]$. Then, by the definition of convexity, there exists a linear function $y = m(x-a) + \varphi(a)$ such that $y \leq \varphi(x)$ for all $x \in \R$, or $m(X-a) + \varphi(a) \leq \varphi(X)$. Taking expectations gives 
    \begin{align}
        \varphi(\E[X]) = \varphi(a) \leq \E[\varphi(X)].
    \end{align}
\end{proof}

\begin{lemma}[\eax{Minkowski's inequality}]
    Let $1 \leq p < \infty$. Then for any $X,Y \in L^{p}$, we have
    \begin{align}
        \norm{X+Y}_{p} \leq \norm{X}_{p} + \norm{Y}_{p}.
    \end{align}.
\end{lemma}
\begin{proof}
    Set $X' = X/\norm{X}_{p}$ and $Y' = Y/\norm{Y}_{p}$. Then $\norm{X'}_{p} = \norm{Y'}_{p} = 1$. Also set $a = \frac{\norm{X}_{p}}{\norm{X}_{p}+\norm{Y}_{p}}$ and $b = \frac{\norm{Y}_{p}}{\norm{X}_{p}+\norm{Y}_{p}}$. Since $x \mapsto x^{p}$ is convex, we have
    \begin{align}
        \abs{aX'+bY'}^{p} \leq a\abs{X'}^{p} + b\abs{Y'}^{p} \implies \E[\abs{aX'+bY'}^{p}] \leq a\E[\abs{X'}^{p}] + b\E[\abs{Y'}^{p}] = a+b = 1.
    \end{align}
    Rearranging gives
    \begin{align}
        \norm{X+Y}_{p} &= (\E[\abs{X+Y}^{p}])^{1/p} = (\E[\abs{(a+b)(X'+Y')}^{p}])^{1/p} \notag \\ &= (a+b)(\E[\abs{aX'+bY'}^{p}])^{1/p} \leq a+b = \norm{X}_{p} + \norm{Y}_{p}.
    \end{align}
\end{proof}

\begin{theorem}[\eax{Hölder's inequality}]
    Let $1 < p,q < \infty$ be such that $\frac{1}{p}+\frac{1}{q} = 1$. Then for any $X \in L^{p}$ and $Y \in L^{q}$, we have $XY \in L^{1}$ and
    \begin{align}
        \E[\abs{XY}] \leq \norm{X}_{p}\norm{Y}_{q}.
    \end{align}
\end{theorem}
Note that the case $p = q = 2$ is a special case of this theorem, and is known as the Cauchy-Schwarz inequality.
\begin{proof}
    We first show that $\frac{a^{p}}{p} + \frac{b^{q}}{q} \geq ab$ for all $a,b > 0$. To see this, note that the function $f(a) = \frac{a^{p}}{p} - ab + \frac{b^{q}}{q}$ is convex in $a$, and has a unique minimum at $a = b^{1/(p-1)}$. Evaluating $f$ at this point gives
    \begin{align}
        f(b^{1/(p-1)}) = \frac{b^{p/(p-1)}}{p} - b^{p/(p-1)} + \frac{b^{q}}{q} = b^{q}\left(\frac{1}{p}-1+\frac{1}{q}\right) = 0.
    \end{align}
    This shows that $\frac{a^{p}}{p} + \frac{b^{q}}{q} \geq ab$ for all $a,b > 0$. Now, let $X \in L^{p}$ and $Y \in L^{q}$, and set $X' = X/\norm{X}_{p}$ and $Y' = Y/\norm{Y}_{q}$. Then $\norm{X'}_{p} = \norm{Y'}_{q} = 1$. Applying the inequality above with $a = \abs{X'}$ and $b = \abs{Y'}$, we have
    \begin{align}
        \abs{X'Y'} \leq \frac{\abs{X'}^{p}}{p} + \frac{\abs{Y'}^{q}}{q} \implies \E[\abs{X'Y'}] \leq \frac{\E[\abs{X'}^{p}]}{p} + \frac{\E[\abs{Y'}^{q}]}{q} = \frac{1}{p} + \frac{1}{q} = 1.
    \end{align}
    Multiplying both sides by $\norm{X}_{p}\norm{Y}_{q}$ gives the desired result.
\end{proof}
With the norm defined on $L^{p}$, we can define convergence in $L^{p}$ by saying that a sequence $(X_{n})_{n \geq 1}$ converges to $X$ in $L^{p}$ if $\norm{X_{n}-X}_{p} \to 0$ as $n \to \infty$.
